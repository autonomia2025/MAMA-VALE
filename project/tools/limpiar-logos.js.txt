// args: jobs, readFile, readImage, saveFile, createCanvas, log
const results = [];
for (const job of jobs) {
  const src = job.src;
  let im; try { im = src.startsWith('assets') ? await readImage(src) : await createImageBitmap(await fetch(src, { mode: 'cors' }).then(r => r.blob())); } catch (e) { try { im = await createImageBitmap(await fetch(src, { mode: 'cors', cache: 'reload' }).then(r => r.blob())); } catch (e2) { results.push(job.id + ': ERROR carga'); continue; } }
  const s0 = Math.min(1, 1000 / Math.max(im.width, im.height));
  const W = Math.round(im.width * s0), H = Math.round(im.height * s0);
  const c = createCanvas(W, H), x = c.getContext('2d', { willReadFrequently: true });
  x.fillStyle = '#fff'; x.fillRect(0, 0, W, H); x.drawImage(im, 0, 0, W, H);
  const img = x.getImageData(0, 0, W, H), px = img.data;
  const at = (xx, yy) => (yy * W + xx) * 4;
  let seed = 13; const rnd = () => { seed = (seed * 16807) % 2147483647; return seed / 2147483647; };
  const info = [];
  for (const r of job.regions) {
    if (r.type === 'h') {
      const w = r.x1 - r.x0;
      const sd = r.sd ?? 2.5;
      for (let yy = r.y0; yy < r.y1; yy++) {
        const A = [0, 1, 2].map(ch => (px[at(r.x0 - 1, yy) + ch] + px[at(r.x0 - 2, yy) + ch] + px[at(r.x0 - 3, yy) + ch]) / 3);
        const B = [0, 1, 2].map(ch => (px[at(r.x1, yy) + ch] + px[at(r.x1 + 1, yy) + ch] + px[at(r.x1 + 2, yy) + ch]) / 3);
        for (let xx = r.x0; xx < r.x1; xx++) { const t = (xx - r.x0 + 1) / (w + 1), nz = (rnd() + rnd() + rnd() - 1.5) * sd * 1.15; for (let ch = 0; ch < 3; ch++) px[at(xx, yy) + ch] = Math.max(0, Math.min(255, Math.round(A[ch] * (1 - t) + B[ch] * t + nz))); }
      }
      info.push('h ' + [r.x0, r.y0, r.x1, r.y1].join(','));
      continue;
    }
    if (r.type === 'v') {
      const w = r.x1 - r.x0, h = r.y1 - r.y0;
      let s = 0, s2 = 0, n = 0; for (let yy = r.y0 - 12; yy < r.y0 - 1; yy++) for (let xx = r.x0; xx < r.x1; xx++) { const dd = px[at(xx, yy + 1) + 2] - px[at(xx, yy) + 2]; s += dd; s2 += dd * dd; n++; }
      const sd = r.sd ?? Math.min(3, Math.sqrt(Math.max(0, s2 / n - (s / n) ** 2)) / Math.SQRT2);
      for (let xx = r.x0; xx < r.x1; xx++) {
        const A = [0, 1, 2].map(ch => (px[at(xx, r.y0 - 1) + ch] + px[at(xx, r.y0 - 2) + ch] + px[at(xx, r.y0 - 3) + ch]) / 3);
        const B = [0, 1, 2].map(ch => (px[at(xx, r.y1) + ch] + px[at(xx, r.y1 + 1) + ch] + px[at(xx, r.y1 + 2) + ch]) / 3);
        for (let yy = r.y0; yy < r.y1; yy++) { const t = (yy - r.y0 + 1) / (h + 1), nz = (rnd() + rnd() + rnd() - 1.5) * sd * 1.15; for (let ch = 0; ch < 3; ch++) px[at(xx, yy) + ch] = Math.max(0, Math.min(255, Math.round(A[ch] * (1 - t) + B[ch] * t + nz))); }
      }
      info.push('v ' + [r.x0, r.y0, r.x1, r.y1].join(',') + ' sd' + sd.toFixed(1));
      continue;
    }
    // mask mode: polygon + pixel selector, harmonic fill
    const P = r.poly; const bx0 = Math.max(1, Math.min(...P.map(p => p[0]))), bx1 = Math.min(W - 2, Math.max(...P.map(p => p[0]))), by0 = Math.max(1, Math.min(...P.map(p => p[1]))), by1 = Math.min(H - 2, Math.max(...P.map(p => p[1])));
    const inPoly = (X, Y) => { let ins = false; for (let i = 0, j = P.length - 1; i < P.length; j = i++) { const [xi, yi] = P[i], [xj, yj] = P[j]; if ((yi > Y) !== (yj > Y) && X < (xj - xi) * (Y - yi) / (yj - yi) + xi) ins = !ins; } return ins; };
    const bw = bx1 - bx0 + 1, bh = by1 - by0 + 1;
    const hsv = (i) => { const R = px[i], G = px[i + 1], B = px[i + 2], mx = Math.max(R, G, B), mn = Math.min(R, G, B); return [mx ? (mx - mn) / mx : 0, mx]; };
    // reference: median sat/val of non-text pixels in polygon
    const ss = [], vv = [];
    for (let yy = by0; yy <= by1; yy++) for (let xx = bx0; xx <= bx1; xx++) if (inPoly(xx, yy)) { const [sa, va] = hsv(at(xx, yy)); ss.push(sa); vv.push(va); }
    ss.sort((a, b) => a - b); vv.sort((a, b) => a - b);
    const Sm = ss[Math.floor(ss.length * .75)], Vlo = vv[Math.floor(vv.length * .3)], Vmed = vv[vv.length >> 1];
    // 'diff': distance from the polygon's median color (logo of any hue on a flat surface)
    let MR = 0, MG = 0, MB = 0; if (r.sel === 'diff') { const cr = [[], [], []]; for (let yy = by0; yy <= by1; yy++) for (let xx = bx0; xx <= bx1; xx++) if (inPoly(xx, yy)) { const i = at(xx, yy); cr[0].push(px[i]); cr[1].push(px[i + 1]); cr[2].push(px[i + 2]); } [MR, MG, MB] = cr.map(a => a.sort((p, q) => p - q)[a.length >> 1]); }
    let m = new Uint8Array(bw * bh);
    for (let yy = by0; yy <= by1; yy++) for (let xx = bx0; xx <= bx1; xx++) if (inPoly(xx, yy)) {
      const [sa, va] = hsv(at(xx, yy)); let hit = false;
      if (r.sel === 'lowsat') hit = sa < Sm * (r.k || .72);
      else if (r.sel === 'bright') hit = va > Vlo + (r.k || 45);
      else if (r.sel === 'white') hit = sa < Sm * (r.k || .6) && va > Vmed * .8;
      else if (r.sel === 'sat') hit = sa > (r.k || .35) && va > 50;
      else if (r.sel === 'dark') hit = va < Vmed - (r.k || 50);
      else if (r.sel === 'yellow') { const i = at(xx, yy); hit = (px[i] + px[i + 1]) / 2 - px[i + 2] > (r.k || 40); }
      else if (r.sel === 'diff') { const i = at(xx, yy); hit = Math.hypot(px[i] - MR, px[i + 1] - MG, px[i + 2] - MB) > (r.k || 45); }
      else hit = true;
      if (hit) m[(yy - by0) * bw + (xx - bx0)] = 1;
    }
    for (let it = 0; it < (r.dil ?? 2); it++) { const nm = m.slice(); for (let yy = 1; yy < bh - 1; yy++) for (let xx = 1; xx < bw - 1; xx++) { const q = yy * bw + xx; if (m[q]) continue; if (m[q - 1] || m[q + 1] || m[q - bw] || m[q + bw]) nm[q] = 1; } m = nm; }
    let cnt = 0; m.forEach(v => cnt += v);
    // grain from unmasked pixels in polygon
    let s = 0, s2 = 0, n = 0; for (let yy = by0; yy <= by1; yy++) for (let xx = bx0; xx < bx1; xx++) { const q = (yy - by0) * bw + (xx - bx0); if (m[q] || m[q + 1] || !inPoly(xx, yy)) continue; const dd = px[at(xx + 1, yy) + 2] - px[at(xx, yy) + 2]; s += dd; s2 += dd * dd; n++; }
    const sd = r.sd ?? Math.min(4, n ? Math.sqrt(Math.max(0, s2 / n - (s / n) ** 2)) / Math.SQRT2 : 1);
    const F = new Float32Array(bw * bh * 3);
    for (let q = 0; q < bw * bh; q++) { const i = at(bx0 + q % bw, by0 + Math.floor(q / bw)); F[q * 3] = px[i]; F[q * 3 + 1] = px[i + 1]; F[q * 3 + 2] = px[i + 2]; }
    // init masked with row-wise interpolation
    for (let yy = 0; yy < bh; yy++) { let xx = 0; while (xx < bw) { if (!m[yy * bw + xx]) { xx++; continue; } const a = xx - 1; let b = xx; while (b < bw && m[yy * bw + b]) b++; const A = a >= 0 ? a : b, B = b < bw ? b : a; for (let k = xx; k < b; k++) { const t = (k - a) / (b - a); for (let ch = 0; ch < 3; ch++) F[(yy * bw + k) * 3 + ch] = F[(yy * bw + A) * 3 + ch] * (1 - t) + F[(yy * bw + B) * 3 + ch] * t; } xx = b; } }
    const iters = r.it || 700;
    for (let it = 0; it < iters; it++) for (let yy = 1; yy < bh - 1; yy++) for (let xx = 1; xx < bw - 1; xx++) { const q = yy * bw + xx; if (!m[q]) continue; for (let ch = 0; ch < 3; ch++) F[q * 3 + ch] = (F[(q - 1) * 3 + ch] + F[(q + 1) * 3 + ch] + F[(q - bw) * 3 + ch] + F[(q + bw) * 3 + ch]) * .25; }
    for (let q = 0; q < bw * bh; q++) if (m[q]) { const i = at(bx0 + q % bw, by0 + Math.floor(q / bw)), nz = (rnd() + rnd() + rnd() - 1.5) * sd * 1.15; for (let ch = 0; ch < 3; ch++) px[i + ch] = Math.max(0, Math.min(255, Math.round(F[q * 3 + ch] + nz))); }
    info.push('mask ' + r.sel + ' px' + cnt + ' sd' + sd.toFixed(1));
  }
  x.putImageData(img, 0, 0);
  await saveFile('assets/kajova/' + job.id + '-limpia.jpg', await c.convertToBlob({ type: 'image/jpeg', quality: .92 }));
  results.push(job.id + ': ' + info.join(' | '));
}
log(results.join('\n'));
